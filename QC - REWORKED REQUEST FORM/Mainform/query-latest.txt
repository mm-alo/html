function num(id) {

    const el = document.getElementById(id);

    if (!el) return 0;

    return parseFloat(String(el.value).replace(/,/g, "")) || 0;
}

function setVal(id, value, decimals) {

    const el = document.getElementById(id);

    if (!el) return;

    el.value = Number(value).toFixed(decimals);
}

function calculateRework() {
    
    //Top section
    const aveCT = num("AVECT");
    const manRate = 75;
    const materialCost = num("reworkmaterialcost");
    const energyCost1 = num("energytxt01");
    const energyCost2 = num("energytxt02");
    const reworkTime = aveCT / 3600;
    const laborCost =
        Number((reworkTime * manRate).toFixed(4));

    const reworkCost =
        Math.ceil(
            (
                laborCost +
                materialCost +
                energyCost1 +
                energyCost2
            ) * 1000
        ) / 1000;
    setVal(
        "REWORKTIME",
        reworkTime,
        4
    );

    setVal(
        "LABORCOST",
        laborCost,
        4
    );

    setVal(
        "REWORKCOST",
        reworkCost,
        4
    );

    setVal(
        "REWORKCOSTPP",
        reworkCost,
        4
    );

    /* ==========================
       BOTTOM TABLE
       ========================== */

    let rawTotalCOPQ = 0;

    for (let i = 1; i <= 10; i++) {

        const row =
            String(i).padStart(2, "0");

       const lotQty =
    num("LOTQTY" + row);

    const numTeeth =
        num("NUMTEETH" + row);

    const copq =
        reworkCost * numTeeth * lotQty;

        rawTotalCOPQ += copq;

        /* COPQ PER LOT */

        setVal(
            "COPQPLOT" + row,
            copq,
            3
        );

        /*
           RIGHTMOST COLUMN

           Excel:
           184.851 -> 185
           224.035 -> 225
           226.002 -> 227

           ROUNDUP
        */

        const roundedCopq =
            Math.ceil(copq);

        if (i < 10) {
        setVal(
            "reworkcostpiece" + row,
            roundedCopq,
            0
        );
    }

    }

    //footer
    setVal(
        "sumCOPQ",
        rawTotalCOPQ,
        3
    );

    const reworkCostPieceTotal =
    document.getElementById("reworkcostpiecetotal");

    if (reworkCostPieceTotal) {
        reworkCostPieceTotal.value =
            "" +
            (Math.ceil(rawTotalCOPQ * 100) / 100)
                .toFixed(2);
    }

    const totalField =
        document.getElementById(
            "TOTALCOPQ"
        );

    if (totalField) {

        totalField.value =
            new Intl.NumberFormat(
                "en-PH",
                {
                    style: "currency",
                    currency: "PHP",
                    minimumFractionDigits: 2,
                    maximumFractionDigits: 2
                }
            ).format(rawTotalCOPQ);
    }
}
